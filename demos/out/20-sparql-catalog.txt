PREFIX t: <urn:ao:term#> PREFIX ca: <urn:ao:column_attribute#> PREFIX cat: <urn:ao:column_attribute__term#> PREFIX cha: <urn:ao:column__has_attribute#> PREFIX col: <urn:ao:catalog_column#> PREFIX tb: <urn:ao:catalog_table#> PREFIX sc: <urn:ao:catalog_schema#> PREFIX db: <urn:ao:catalog_database#> SELECT ?term ?attr ?column ?table ?schema ?database WHERE { GRAPH ?g {
  ?tl cat:ref-term_id ?t ; cat:ref-attribute_id ?a .
  ?t t:name ?term . ?a ca:name ?attr .
  ?hl cha:ref-attribute_id ?a ; cha:ref-column_id ?c .
  ?c col:name ?column ; col:ref-table_id ?tbl .
  ?tbl tb:name ?table ; tb:ref-schema_id ?s .
  ?s sc:name ?schema ; sc:ref-database_id ?d . ?d db:name ?database .
} } ORDER BY ?term ?attr
